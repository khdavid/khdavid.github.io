#include <Wire.h>
#include <ITG3200.h>
#include <Servo.h> 
ITG3200 gyro = ITG3200();
float  x,y,z;
float angle=0;
int ix, iy, iz;
unsigned long time_1=0;
unsigned long time_2=0;
Servo myservo; 
int ServoPosition;
void setup(void) 
{
  Serial.begin(9600);
  myservo.attach(4);
  Wire.begin();      
  delay(100);
  gyro.init(ITG3200_ADDR_AD0_HIGH); 
  Serial.print("zeroCalibrating...");
  gyro.zeroCalibrate(2500, 2);//(number of mesaurments, milliseconds between them)
  Serial.println("done.");
}

void loop(void) 
{
  while (gyro.isRawDataReady()) 
  {
    //Reads calibrated values in deg/sec    
    gyro.readGyro(&x,&y,&z); 
    time_1=time_2;
    time_2 = micros();
    if (time_1!=0)
	{
      angle=angle+2*x*((float)time_2-(float)time_1)/100000;  
    }
    // Serial.println(angle);  
    ServoPosition=90-angle;
    myservo.write(ServoPosition);
  } 
}
